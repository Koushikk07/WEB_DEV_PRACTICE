const mongoose = require("mongoose");

const chatSchema = new mongoose.Schema({
  from: {
    type: String,
  },
  to: {
    type: String,
  },
  msg: {
    type: String,
    maxLength: 50,
  },
  created_at: {
    type: Date,
    required: true,
  },
});

const Chat = mongoose.model("Chat", chatSchema);
module.exports = Chat;
